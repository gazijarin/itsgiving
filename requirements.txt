# Verified on Python 3.12 / macOS arm64: `pip install -r requirements.txt` and run.
#
# Don't casually unpin these. mediapipe 0.10.30+ (including 1.0.x) ships macOS
# wheels that abort() on startup — "Check failed: service_ Service is unavailable",
# or a MetalHelper crash — so mediapipe is held at the last build that works.
#
# That build needs numpy<2, and opencv 5 needs numpy>=2. mediapipe 0.10.21 also
# asks for an *unpinned* opencv-contrib-python, which is what quietly drags in
# opencv 5 and therefore numpy 2 and therefore a broken install. The explicit
# opencv pins below are what stop that. Unpin one and you have to unpin all three.
mediapipe==0.10.21
numpy<2
opencv-python<5
opencv-contrib-python<5
protobuf>=4.25.3,<5
pyvirtualcam
pillow
pygame-ce
